# Requirements for the Hugging Face Gradio SDK Space (uploaded as requirements.txt at the
# Space root). Linux-only: the `+cpu` torch pin has no macOS wheel — use the repo's
# requirements.txt for local development.
#
# Only what app.py's import graph needs (app.py -> src.predict -> src.config).
# Deliberately absent: pandas, scikit-learn, scipy, matplotlib, seaborn, jupyter, nbconvert,
# iterative-stratification. Verified by installing exactly this set into a clean venv.
# gradio already brings fastapi/uvicorn, so those are not listed.

# CPU-only torch: the default PyPI Linux wheel bundles CUDA (~2.5GB) and would blow up the
# Space build. The `+cpu` local version is required — with --extra-index-url alone, pip sees
# both 2.13.0 (CUDA) and 2.13.0+cpu and may pick either.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.13.0+cpu

transformers==5.14.1
numpy==2.5.1
gradio==6.20.0
